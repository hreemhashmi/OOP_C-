using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace RMS_project.Dish
{
    internal class DishRepo
    {
        private readonly string file = "dish.txt";
        public DishRepo() {
            if (!File.Exists(file))
            {
                SaveInFile(new DishModel("biryani", 270));
                SaveInFile(new DishModel("burger", 200));
                SaveInFile(new DishModel("juice", 120));
                SaveInFile(new DishModel("pizza", 550));
            }

        }
        public void SaveInFile(DishModel dish)
        {
            using (StreamWriter stream = new StreamWriter(file, true))
            {
                stream.WriteLine(dish.ToString());
            }
        }
        public List<DishModel> GetDishesFromFiles()
        {
            List<DishModel> dishModels = new List<DishModel>();
            if (!File.Exists(file))
            {
                return dishModels;
            }
            using (StreamReader stream = new StreamReader(file))
            {
                string line = "";
                while ((line = stream.ReadLine()) != null)
                {
                    string[] parts = line.Split(',');
                    string dishname = parts[0];
                    double price = double.Parse(parts[1]);
                    DishModel dishModel = new DishModel(dishname,price);
                    dishModels.Add(dishModel);
                }
            }
            return dishModels;
        }
        public void SaveAllInFile(List<DishModel> dishes)
        {
            using (StreamWriter stream = new StreamWriter(file, false))
            {
                for (int i = 0; i < dishes.Count; i++)
                {
                    stream.WriteLine(dishes[i].ToString());
                }

            }
        }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace RMS_project.Dish
{
    internal class DishService
    {
      
        private DishRepo repo = new DishRepo();

        public void AddDish(DishModel dish)
        {
            repo.SaveInFile(dish);
        }

        public List<DishModel> GetAllDishes()
        {
            return repo.GetDishesFromFiles();
        }

        public DishModel SearchDish(string name)
        {
            List<DishModel> dishes = repo.GetDishesFromFiles();
            for (int i = 0; i < dishes.Count; i++)
            {
                if (dishes[i].GetName() == name)
                {
                    return dishes[i];
                }
            }
            return null;
        }

        public bool UpdateDish(string oldName, string newName, float newPrice)
        {
            List<DishModel> dishes = repo.GetDishesFromFiles();
            for (int i = 0; i < dishes.Count; i++)
            {
                if (dishes[i].GetName() == oldName)
                {
                    dishes[i].SetName(newName);
                    dishes[i].SetPrice(newPrice);
                    repo.SaveAllInFile(dishes);
                    return true;
                }
            }
            return false;
        }

        public bool DeleteDish(string name)
        {
            List<DishModel> dishes = repo.GetDishesFromFiles();
            List<DishModel> remaining = new List<DishModel>();
            bool found = false;

            for (int i = 0; i < dishes.Count; i++)
            {
                if (dishes[i].GetName() == name)
                {
                    found = true;
                }
                else
                {
                    remaining.Add(dishes[i]);
                }
            }

            if (found == true)
            {
                repo.SaveAllInFile(remaining);
            }
            return found;
        }

        public void SortDishes(bool lowToHigh)
        {
            List<DishModel> dishes = repo.GetDishesFromFiles();

            for (int i = 0; i < dishes.Count; i++)
            {
                for (int j = i + 1; j < dishes.Count; j++)
                {
                    bool swap = false;
                    if (lowToHigh == true && dishes[i].GetPrice() > dishes[j].GetPrice())
                    {
                        swap = true;
                    }
                    if (lowToHigh == false && dishes[i].GetPrice() < dishes[j].GetPrice())
                    {
                        swap = true;
                    }

                    if (swap == true)
                    {
                        DishModel temp = dishes[i];
                        dishes[i] = dishes[j];
                        dishes[j] = temp;
                    }
                }
            }
            repo.SaveAllInFile(dishes);
        }
    }
}

using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace RMS_project.Order
{
    internal class OrderRepo
    {

        private readonly string file = "orders.txt";

        public OrderRepo() { }

        public void SaveInFile(OrderModel order)
        {
            using (StreamWriter stream = new StreamWriter(file, true))
            {
                stream.WriteLine(order.ToString());
            }
        }

        public List<OrderModel> GetOrdersFromFile()
        {
            List<OrderModel> orderModels = new List<OrderModel>();

            if (!File.Exists(file))
            {
                return orderModels;
            }

            using (StreamReader stream = new StreamReader(file))
            {
                string line = "";
                while ((line = stream.ReadLine()) != null)
                {
                    string[] parts = line.Split(',');
                    string dishName = parts[0];
                    int quantity = int.Parse(parts[1]);
                    double total = double.Parse(parts[2]);
                    string status = parts[3];

                    OrderModel orderModel = new OrderModel(dishName, quantity, total, status);
                    orderModels.Add(orderModel);
                }
            }
            return orderModels;
        }

        public void SaveAllInFile(List<OrderModel> orders)
        {
            using (StreamWriter stream = new StreamWriter(file, false))
            {
                for (int i = 0; i < orders.Count; i++)
                {
                    stream.WriteLine(orders[i].ToString());
                }
            }
        }
    }
}

using RMS_project.Dish;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace RMS_project.Order
{
    internal class OrderService
    {
        private OrderRepo repo = new OrderRepo();
        private DishService dishService = new DishService();

        public bool PlaceOrder(string dishName, int quantity)
        {
            DishModel dish = dishService.SearchDish(dishName);
            if (dish == null || quantity <= 0)
            {
                return false;
            }
            double total = dish.GetPrice() * quantity;
            OrderModel order = new OrderModel(dishName, quantity, total, "pending");
            repo.SaveInFile(order);
            return true;
        }

        public List<OrderModel> GetAllOrders()
        {
            return repo.GetOrdersFromFile();
        }

        public List<OrderModel> GetCurrentOrders()
        {
            List<OrderModel> all = repo.GetOrdersFromFile();
            List<OrderModel> current = new List<OrderModel>();
            for (int i = 0; i < all.Count; i++)
            {
                string status = all[i].GetStatus();
                if (status == "pending" || status == "confirmed")
                {
                    current.Add(all[i]);
                }
            }
            return current;
        }

        public bool RemoveOrder(string dishName)
        {
            List<OrderModel> orders = repo.GetOrdersFromFile();
            List<OrderModel> remaining = new List<OrderModel>();
            bool found = false;

            for (int i = 0; i < orders.Count; i++)
            {
                if (found == false && orders[i].GetDishName() == dishName && orders[i].GetStatus() == "pending")
                {
                    found = true;
                }
                else
                {
                    remaining.Add(orders[i]);
                }
            }
            if (found == true)
            {
                repo.SaveAllInFile(remaining);
            }
            return found;
        }

        public bool UpdateQuantity(string dishName, int newQuantity)
        {
            DishModel dish = dishService.SearchDish(dishName);
            if (dish == null || newQuantity <= 0)
            {
                return false;
            }
            List<OrderModel> orders = repo.GetOrdersFromFile();
            for (int i = 0; i < orders.Count; i++)
            {
                if (orders[i].GetDishName() == dishName && orders[i].GetStatus() == "pending")
                {
                    orders[i].SetQuantity(newQuantity);
                    orders[i].SetTotal(dish.GetPrice() * newQuantity);
                    repo.SaveAllInFile(orders);
                    return true;
                }
            }
            return false;
        }

        public bool ChangeAllPending(string newStatus)
        {
            List<OrderModel> orders = repo.GetOrdersFromFile();
            bool found = false;
            for (int i = 0; i < orders.Count; i++)
            {
                if (orders[i].GetStatus() == "pending")
                {
                    orders[i].SetStatus(newStatus);
                    found = true;
                }
            }
            if (found == true)
            {
                repo.SaveAllInFile(orders);
            }
            return found;
        }

        public double CalculateBill()
        {
            List<OrderModel> current = GetCurrentOrders();
            double bill = 0;
            for (int i = 0; i < current.Count; i++)
            {
                bill = bill + current[i].GetTotal();
            }
            return bill;
        }

        public double GetDiscountPercent(double bill)
        {
            if (bill >= 5000)
            {
                return 10;
            }
            else if (bill >= 1000)
            {
                return 5;
            }
            return 0;
        }

        public double GetCouponPercent(string code)
        {
            if (code == "hm207")
            {
                return 7;
            }
            return 0;
        }
        public bool UpdateStatus(string dishName, string newStatus)
        {
            List<OrderModel> orders = repo.GetOrdersFromFile();
            for (int i = 0; i < orders.Count; i++)
            {
                string status = orders[i].GetStatus();
                if (orders[i].GetDishName() == dishName && status != "canceled" && status != "completed")
                {
                    orders[i].SetStatus(newStatus);
                    repo.SaveAllInFile(orders);
                    return true;
                }
            }
            return false;
        }

        public List<OrderModel> GetOrdersByStatus(string status)
        {
            List<OrderModel> all = repo.GetOrdersFromFile();
            List<OrderModel> result = new List<OrderModel>();
            for (int i = 0; i < all.Count; i++)
            {
                if (all[i].GetStatus() == status)
                {
                    result.Add(all[i]);
                }
            }
            return result;
        }

        public double GetTotalRevenue()
        {
            List<OrderModel> orders = repo.GetOrdersFromFile();
            double revenue = 0;
            for (int i = 0; i < orders.Count; i++)
            {
                if (orders[i].GetStatus() != "canceled")
                {
                    revenue = revenue + orders[i].GetTotal();
                }
            }
            return revenue;
        }

    }
